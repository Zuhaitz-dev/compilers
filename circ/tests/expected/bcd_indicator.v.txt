module circuit (
  input CLK,
  output Q0,
  output Q1,
  output Q2,
  output Q3,
  output RST
);
  wire NOT_0;
  reg DFF_1 = 1'b0;
  wire NOT_2;
  reg DFF_3 = 1'b0;
  wire NOT_4;
  reg DFF_5 = 1'b0;
  wire NOT_6;
  reg DFF_7 = 1'b0;
  wire AND_8;
  always @(posedge CLK) DFF_1 <= NOT_0;
  assign NOT_0 = ~DFF_1;
  always @(posedge NOT_0) DFF_3 <= NOT_2;
  assign NOT_2 = ~DFF_3;
  always @(posedge NOT_2) DFF_5 <= NOT_4;
  assign NOT_4 = ~DFF_5;
  always @(posedge NOT_4) DFF_7 <= NOT_6;
  assign NOT_6 = ~DFF_7;
  assign AND_8 = DFF_7 & DFF_3;
  assign Q0 = DFF_1;
  assign Q1 = DFF_3;
  assign Q2 = DFF_5;
  assign Q3 = DFF_7;
  assign RST = AND_8;
endmodule
