module full_adder (
  input A
  , input B
  , input Cin
  , output S
  , output Cout
);
  wire XOR_0;
  wire XOR_1;
  wire AND_2;
  wire AND_3;
  wire OR_4;
  assign XOR_0 = A ^ B;
  assign XOR_1 = XOR_0 ^ Cin;
  assign AND_2 = A & B;
  assign AND_3 = XOR_0 & Cin;
  assign OR_4 = AND_2 | AND_3;
  assign S = XOR_1;
  assign Cout = OR_4;
endmodule

module circuit (
  input A0,
  input B0,
  input Cin,
  input A1,
  input B1,
  output S0,
  output S1
);
  wire _w0;
  full_adder full_adder_0 (
    .A(A0),
    .B(B0),
    .Cin(Cin),
    .S(S0),
    .Cout(_w0)
  );
  full_adder full_adder_1 (
    .A(A1),
    .B(B1),
    .Cin(_w0),
    .S(S1),
    .Cout(/*unused*/)
  );
endmodule
