module circuit (
  input [3:0] A,
  input [3:0] B,
  input OP,
  input CLK,
  output [3:0] Q
);
  wire [3:0] AND_0;
  wire [3:0] XOR_1;
  wire NOT_2;
  wire [3:0] AND_3;
  wire [3:0] AND_4;
  wire [3:0] OR_5;
  reg [3:0] DFF_6 = 1'b0;
  assign AND_0 = A & B;
  assign XOR_1 = A ^ B;
  assign NOT_2 = ~OP;
  assign AND_3 = AND_0 & {4{NOT_2}};
  assign AND_4 = XOR_1 & {4{OP}};
  assign OR_5 = AND_3 | AND_4;
  always @(posedge CLK) DFF_6 <= OR_5;
  assign Q = DFF_6;
endmodule
