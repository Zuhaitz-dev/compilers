#include <stdint.h>

void circuit(int S, int A, int B, int *Q) {
    int NOT_0;
    int AND_1;
    int AND_2;
    int OR_3;
    int _changed = 1, _guard = 0;
    static int _inited = 0;
    if (!_inited) {
    NOT_0 = (~0) & 0x1;
    AND_1 = 0 & (0 * 0xF);
    AND_2 = 0 & (NOT_0 * 0xF);
    OR_3 = AND_1 | AND_2;
        _inited = 1;
    }
    while (_changed && _guard++ < 15) {
        _changed = 0;
    { int _t = (~S) & 0x1; if (_t != NOT_0) { NOT_0 = _t; _changed = 1; } }
    { int _t = A & (S * 0xF); if (_t != AND_1) { AND_1 = _t; _changed = 1; } }
    { int _t = B & (NOT_0 * 0xF); if (_t != AND_2) { AND_2 = _t; _changed = 1; } }
    { int _t = AND_1 | AND_2; if (_t != OR_3) { OR_3 = _t; _changed = 1; } }
    if (Q) *Q = OR_3;
    }
}
